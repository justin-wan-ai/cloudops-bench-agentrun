# AgentRun DSL v0.4 -- agentic root cause analysis over a frozen incident snapshot
#
# v0.4 = v0.3 behaviour, with ONE change: every branch decision that used to live in
# Python (`plans.FAMILY_PLANS`, `plans.shortlist` ordering, the inline `when` lambdas,
# the implicit "fall back to runtime" default) is now declared here.
#
#   DSL  declares : which nodes run, in what order, which probe sequence each family
#                   runs, which gate guards each probe, how the mechanical shortlist is
#                   ordered, and which family is the fallback.
#   code implements: the semantics of the named probes / gates / signals (kubectl
#                   syntax, regexes), the rule engine, and node dispatch.
#
# Consequence: deleting `jev: route`, a `family_plan` entry, or renaming a gate is a
# LOAD-TIME error (dsl_parse._validate_policy) instead of a silent behaviour change.
#
# Semantics implemented by agentrun/runner.py:
#   * node PRESENCE decides whether a step runs at all;
#   * execution order is fixed (call -> derive -> pick -> verify); the DSL selects the
#     chain, the loop bound, the plan tables and the gates -- not line-by-line execution;
#   * every parameter below must be able to change behaviour (P5 gate).

dsl: agentrun-rca
version: "0.4"

# ---------------------------------------------------------------- scalar knobs
params:
  cands_cap: 8              # candidates handed to jev(pick)
  max_waves: 5              # bounded loop (one probe per wave)
  min_waves: 5              # probe are free: never stop before the plan is exhausted
  steps_per_wave: 1         # one probe per wave, so wave N+1 can gate on wave N
  dominance_ratio: 1.5      # mechanical pick only when top score dominates the runner-up
  conf_min: 0.0             # absolute floor on P(supported); 0 = only require beating alternatives
  escalate: true
  escalate_budget: 8        # judge agent tool-call budget
  judge_on_codedefect: true # code defects stay open-ended until code evidence pins them
  jev_route: true
  jev_pick: true
  jev_verify: true

# ---------------------------------------------------------------- route policy
# Mechanical shortlist, evaluated top-down. The first match wins; the order is what
# decides "which family do we try when the model abstains". `residual` always matches,
# so infrastructure is always the last hypothesis.
route_priority:
  - {signal: pending_pods,        family: scheduling}
  - {signal: image_pull,          family: startup}
  - {signal: not_ready_or_crash,  family: runtime}
  - {signal: no_pods_created,     family: admission}
  - {signal: code_shaped_alert,   family: codedefect}
  - {signal: saturation_alert,    family: performance}
  - {signal: healthy_unreachable, family: service}
  - {signal: healthy_degraded,    family: performance}
  - {signal: residual,            family: infrastructure}

# Used only when no signal matches at all (kept explicit so nothing is implicit).
route_fallback: runtime

# ---------------------------------------------------------------- probe plans
# Gate vocabulary (implemented in plans.GATES):
#   always, healthy_system, weak, weak_strong, weak_very_strong,
#   probe_evidence, restart_evidence, probe_or_restart_or_weak,
#   restart_or_weak_strong, volume_evidence
# Probe vocabulary (implemented in plans.PROBES): see dsl_versions/README.md

# wave 0 census: two probes that frame every incident. `alerts` only earns admissible
# evidence when no pod-level fault is visible, so it is gated on a healthy census.
census:
  - {step: pods,   gate: always}
  - {step: alerts, gate: healthy_system}

family_plan:
  admission:
    - {step: deployments,        gate: always}
    - {step: events,             gate: always}          # quota/RBAC rejection lives here
    - {step: resourcequota,      gate: weak_strong}
    - {step: describe_deployment, gate: weak_very_strong}
  scheduling:
    - {step: describe_pod,       gate: always}          # scheduler events + pod spec
    - {step: pvcs,               gate: volume_evidence}
    - {step: cluster_config,     gate: always}          # Pending == capacity/constraint question
  startup:
    - {step: describe_pod,       gate: always}
    - {step: app_yaml,           gate: weak_strong}
    - {step: error_logs,         gate: weak}
  runtime:
    - {step: describe_pod,       gate: always}          # probe config, state, restarts
    - {step: app_yaml,           gate: probe_or_restart_or_weak}
    - {step: error_logs,         gate: restart_or_weak_strong}
  service:
    - {step: services,           gate: always}
    - {step: endpoints,          gate: always}
    - {step: app_yaml,           gate: weak}
    - {step: connectivity,       gate: weak_strong}
    - {step: deps,               gate: weak_strong}
  performance:
    - {step: recent_logs,        gate: always}
    - {step: describe_pod,       gate: weak}
    - {step: deps,               gate: weak}
  infrastructure:
    - {step: node_status,        gate: always}
    - {step: cluster_config,     gate: weak}
  codedefect:
    - {step: code_files,         gate: always}
    - {step: source_code,        gate: always}
    - {step: recent_logs,        gate: weak_strong}

# ---------------------------------------------------------------- workflow
chain:
  - code: extract_anchors
  - call: probe_batch         # wave 0 = census declared above
  - jev: route                # restricted choice over the route_priority shortlist
  - code: build_plan          # compiles family_plan[state.family]
  - loop:
      until: solved_or_exhausted
      max_waves: 5
      body:
        - call: probe_batch
        - code: derive_candidates
        - jev: pick
        - jev: verify
  - escalate:
      predicate: no_candidates | contradicted | family_unknown | weak_after_waves
  - code: normalize_answer
  - artifact: persist_trace
